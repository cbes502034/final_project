# --- Web framework ---
# fastapi 0.137+ no longer lists routes added by include_router in app.routes.
# app/ownership.py (progress table) and tests/routes (the "is it done yet" check)
# both read app.routes, so a fresh install would show 0/70 and 85 failing tests.
# Fix those two places before raising this upper bound.
fastapi>=0.115,<0.137
uvicorn[standard]>=0.34,<0.40
pydantic>=2.10,<3
pydantic-settings>=2.6,<3

# --- Database ---
sqlalchemy>=2.0,<3

psycopg[binary]>=3.2,<4

# --- Auth ---
pyjwt>=2.10,<3
# Do not switch back to passlib. It has not been updated since 2020, and its
# backend self-test hashes a sample longer than 72 bytes; bcrypt 4.1+ raises on
# that, so every route that touches a password returns 500 on a fresh install.
bcrypt>=4.1,<6
python-multipart>=0.0.20          # form login

# Training packages (transformers / peft / bitsandbytes / trl) are left out on
# purpose: they are only used in the Colab training notebook, and installing them
